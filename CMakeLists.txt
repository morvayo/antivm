cmake_minimum_required(VERSION 3.20)
project(antivm LANGUAGES CXX)

add_library(antivm INTERFACE)
target_include_directories(antivm INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_compile_features(antivm INTERFACE cxx_std_17)

include(CTest)
if(BUILD_TESTING)
    if(NOT WIN32)
        message(FATAL_ERROR "tests require windows")
    endif()
    foreach(name IN ITEMS vendor formats collected drivers boot trap)
        add_executable(test_${name} tests/${name}.cc)
        target_link_libraries(test_${name} PRIVATE antivm)
        target_include_directories(test_${name} PRIVATE src/unknown)
        if(MSVC)
            target_compile_options(test_${name} PRIVATE /utf-8 /UNDEBUG)
        else()
            target_compile_options(test_${name} PRIVATE -UNDEBUG)
        endif()
        add_test(NAME ${name} COMMAND test_${name})
        set_tests_properties(${name} PROPERTIES TIMEOUT 90)
    endforeach()
    target_sources(test_trap PRIVATE tests/trap_peer.cc)
endif()
